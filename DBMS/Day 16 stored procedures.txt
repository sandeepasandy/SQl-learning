IN Parameter:
AN IN parameter is used to pass a value into a stored procedure

The procedure can use the value, but changes  

OUT Parameter:
An OUT  parameter is used to return  a value

3. OUT PARAMETER:

mysql> DELIMITER //
mysql> CREATE PROCEDURE TOTAL_EMPLOYEES(OUT TOTAL INT)
    -> BEGIN
    -> SELECT COUNT(*)INTO TOTAL FROM EMPLOYEES;
    -> END //
Query OK, 0 rows affected (0.03 sec)
mysql> DELIMITER ;
mysql> SET @TOTAL = 0;
Query OK, 0 rows affected (0.01 sec)

mysql> CALL TOTAL_EMPLOYEES(@TOTAL);
Query OK, 1 row affected (0.02 sec)

mysql> SELECT @TOTAL;
+--------+
| @TOTAL |
+--------+
|     10 |
+--------+

mysql> DELIMITER //
mysql> CREATE PROCEDURE MAXIMUM_SALARY(OUT MAX_SAL DECIMAL(10,2))
    -> BEGIN
    -> SELECT MAX(SALARY) INTO MAX_SAL FROM EMPLOYEES;
    -> END //
Query OK, 0 rows affected (0.02 sec)

mysql> DELIMITER ;
mysql> SET @MAX_SAL=0;
Query OK, 0 rows affected (0.00 sec)

mysql> CALL MAXIMUM_SALARY(@MAX_SAL);
Query OK, 1 row affected (0.01 sec)

mysql> SELECT @MAX_SAL;
+----------+
| @MAX_SAL |
+----------+
| 70000.00 |
+----------+

4. IN AND OUT PARAMETERS:

mysql> DELIMITER //
mysql> CREATE PROCEDURE EMPLOYEE_SALARY(IN EID INT,OUT EMP_SAL DECIMAL(10,2))
    -> BEGIN
    -> SELECT SALARY INTO EMP_SAL FROM EMPLOYEES WHERE EMP_ID=EID;
    -> END //
Query OK, 0 rows affected (0.03 sec)

mysql> DELIMITER ;
mysql> SET @EMP_SALARY=0;
Query OK, 0 rows affected (0.00 sec)

mysql> CALL EMPLOYEE_SALARY(105,@EMP_SALARY);
Query OK, 1 row affected (0.01 sec)

mysql> SELECT @EMP_SALARY;
+-------------+
| @EMP_SALARY |
+-------------+
|    60000.00 |
+-------------+
